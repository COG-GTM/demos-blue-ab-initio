def derive_asset_class(product_col: str) -> Column:
    expr = None
    for asset_class, products in ASSET_CLASS_RULES:
        cond = F.col(product_col).isin(products)
        expr = F.when(cond, asset_class) if expr is None else expr.when(cond, asset_class)
    return expr.otherwise("OTHER")

def fx_lookup(trades: DataFrame, fx_rates: DataFrame) -> DataFrame:
    """lookup on notional_ccy = base_ccy, then the graph's three-way GBP rule."""
    dupes = fx_rates.groupBy("base_ccy").count().filter("count > 1").count()
    if dupes:
        raise ValueError(f"fx lookup key base_ccy is not unique ({dupes} duplicate keys)")
    rates = fx_rates.select(
        F.col("base_ccy").alias("_lk_ccy"), F.trim("spot_rate").cast(FX_RATE).alias("_lk_rate")
    )
    j = trades.join(F.broadcast(rates), trades.notional_ccy == rates._lk_ccy, "left")
    is_gbp = F.col("notional_ccy") == "GBP"
    has_rate = F.col("_lk_rate").isNotNull()

    def to_gbp(amount: str) -> Column:
        return (
            F.when(is_gbp, F.col(amount))
            .when(has_rate, (F.col(amount) * F.col("_lk_rate")).cast(DECIMAL_AMOUNT))
            .otherwise(F.col(amount))
        )

    j = j.withColumn("notional_amount_gbp", to_gbp("notional_amount")).withColumn(
        "market_value_gbp", to_gbp("market_value")
    )
    return j.select(*field_names(TRADE_REF))

    """partition_by_expression: one output flow per source_system expression."""

def load_staging_table(spark: SparkSession, df: DataFrame, table_dir: Path, business_date: str,
                       batch_id: str) -> int:
    """pre_sql DELETE ... WHERE TRADE_DATE = :business_date AND BATCH_ID = :batch_id, then INSERT."""
    new_rows = staging_columns(df, batch_id)
    if table_dir.exists():
        deleted = (F.col("TRADE_DATE") == F.to_date(F.lit(business_date))) & (F.col("BATCH_ID") == int(batch_id))
        kept = spark.read.parquet(str(table_dir)).filter(~F.coalesce(deleted, F.lit(False)))
        combined = kept.unionByName(new_rows)
    else:
        combined = new_rows
    tmp = table_dir.with_name(table_dir.name + ".tmp")
    shutil.rmtree(tmp, ignore_errors=True)
    combined.write.parquet(str(tmp), mode="overwrite")
    shutil.rmtree(table_dir, ignore_errors=True)
    tmp.rename(table_dir)
    spark.catalog.refreshByPath(str(table_dir))
    return spark.read.parquet(str(table_dir)).count()

6 · Legacy behaviours surfaced by the port

The PySpark code reproduces these as-is (so outputs line up); each is a decision for the migration owner. Observations are from this run.

asset_class only derived for Calypso

normalise_murex and normalise_summit never assign out.asset_class, so it is NULL for those feeds.

observed: 60 enriched rows with empty asset_class (MUREX 35, SUMMIT 25)

Currencies without an FX rate pass through unconverted

fx_lookup's else-branch copies the local amount into *_gbp when no rate is found.

observed: 14 rows took the no-rate branch (synthetic SGD trades)

pre_sql DELETE is not rerun-safe

The staging DELETE filters on TRADE_DATE = business date, but most trades have earlier trade dates, so a rerun of the same batch re-inserts them.

observed: staged rows after run 1: 100; after rerun of the same batch: 167 (33 of 100 trades have trade_date = 2026-11-13)

FEED_ID mapped from a field that does not exist

load_calypso_stg maps FEED_ID = CAST(feed_id AS BIGINT) but trade_record_t has no feed_id.

observed: 66 STG_TRADES_CALYPSO rows with NULL FEED_ID (all rows in the table after the same-batch rerun)

Counterparty lookup is read but unused

read_cpty_lookup has no downstream flow in txfm_trade_enrichment.mp.

observed: 12 rows read, 0 consumed

market_value converted at the notional currency rate

The lookup key is notional_ccy, so market_value_gbp uses that rate even if market_value_ccy differs.

observed: 0 rows where market_value_ccy differs from notional_ccy (the synthetic sample does not exercise this case)

What this does and does not show

All inputs are synthetic (calypso_trades_2026-11-13.dat (40 rows), murex_trades_2026-11-13.dat (35 rows), summit_trades_2026-11-13.dat (25 rows), fx_rates_2026-11-13.dat (5 rows), counterparty_master.dat (12 rows)), generated by scripts/generate_synthetic_data.py from the repo's DML. No Ab Initio Co>Operating System or Teradata was available, so no output from the legacy runtime was observed: "parity" here means the PySpark port agrees with an independent re-implementation of the same rules, plus row-count / control-total reconciliation. Teradata staging tables are emulated as local Parquet, and the SP_TRADE_LIFECYCLE stored-procedure phase is not migrated.
